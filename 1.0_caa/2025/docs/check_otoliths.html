checko<- bio |>  
    mutate(region= if_else(nafo %in% c("4T", "4S"), "Gulf",
                                 if_else(prov=="NL" | nafo =="4R", "NL",
                                           if_else(prov=="NS" | nafo %in% c("4X", "4V"), "NS", NA)))) |> 

    filter(agef %in% c(0,1)) |> mutate(Q=quarter(date))

2 Plot N bio data

2.1 Overall age 1 or 0

checkoa = checko |> 
    group_by(year) |>  tally()  |> 
    mutate(`check N > 50`= if_else(n>50,"\u2713", "X")) #|>  pivot_wider(values_from=n, names_from=region)

checkoa |>  ggplot(aes(x=year, y=n, fill=`check N > 50`)) +
    geom_bar(stat="identity", alpha=1, col="black",linewidth=0.1) + 
       scale_fill_manual(values=c("black", "grey"))+
    theme(legend.position="top")

2.2 By age

checko1 = checko |>
