ot <- read_excel("S:/Pélagiques/Maquereau/Otolithes_lectures d'âge/Otholite de référence_lectures.xlsx", sheet="Lectures_MB")

group.gear<- read.delim("../../data/caa_group_gear.txt") %>%  dplyr::rename(gear_name=gear.cat)
group.region<- read.delim("../../data/caa_group_region.txt")



val<- ot %>%  pivot_longer(`2022`:`2026`) %>%  
    mutate(value = if_else(value >10, 10, value),
           age_f=if_else(age_f >10, 10, age_f),
           valid= if_else(age_f == value, "1", "0"),
           diff= value-age_f) %>%  filter(!is.na(age_f)) %>%  
    mutate(nafo=substring(subdivision, 1,2)) # criterias to filter


val<- list(val, group.gear, group.region) %>%  purrr::reduce(left_join)

conc<- val %>%
  dplyr::group_by(name) %>% filter(!is.na(age_f), !is.na(value)) %>% 
  dplyr::summarise(
    n = dplyr::n(),
    `% conc.`= mean(valid == "1", na.rm=T) * 100,
    mean_age_diff=round(mean(diff, na.rm=T),2),
    .groups = "drop"
  )
